## 2. 실행해서 채울 것 (한 번에)

서버·Worker·Docker가 켜진 상태에서 실행한다.

```bash
cd ~/fastapi_resnet18 && source .venv/bin/activate

# (1) 성능 측정 → 산출물 6, 8
python -m tools.measure_timings --repeat 5
lscpu | grep "Model name"; lsb_release -d

# (2) 오류 응답 → 산출물 3, 7 (T-07, T-18, T-19)
curl -s -w "  [%{http_code}]\n" http://127.0.0.1:8000/health
echo bad > /tmp/bad.txt
curl -s -w "  [%{http_code}]\n" -X POST -F "image=@/tmp/bad.txt" http://127.0.0.1:8000/api/sync/classify/
head -c 5242881 /dev/urandom > /tmp/big.jpg
curl -s -w "  [%{http_code}]\n" -X POST -F "image=@/tmp/big.jpg" http://127.0.0.1:8000/api/sync/classify/
curl -s -w "  [%{http_code}]\n" http://127.0.0.1:8000/api/async/jobs/999999/
IMG=$(find dataset/test/healthy -type f | head -1)
curl -s -w "  [%{http_code}]\n" -X POST -F "image=@$IMG" http://127.0.0.1:8000/api/async/classify/

# (3) DB 조회 → 산출물 4, 7 (T-17)
docker compose exec postgres psql -U vision_user -d vision_db -c \
  'SELECT id, status, top1_label, round(top1_score::numeric,4) AS score, inference_ms, cache_hit, created_at FROM image_jobs ORDER BY id;'

# (4) 테스트 → 산출물 7 (T-01, T-20)
python -m pytest -q tests/test_api.py

# (5) 데이터 확인 → 산출물 1
for c in angular_leaf_spot bean_rust healthy; do echo "test/$c: $(find dataset/test/$c -type f | wc -l)"; done
find dataset -type f -exec md5sum {} + | sort | uniq -w32 -D | head
ls -lh artifacts/beans_resnet18/
history | grep "finetune.*e30"
```
